   147→  const map: Record<
   148→    string,

   153→      icon: Signal,
   154→      text: `${t.koneksiMode}: …`,
   155→    },
   156→    socket: {
   157→      cls: "bg-emerald-950 text-emerald-400 border-emerald-900",
   158→      icon: SignalHigh,
   159→      text: `${t.koneksiSoket} ${t.terhubung}`,

   162→      cls: "bg-amber-950 text-amber-400 border-amber-900",
   163→      icon: Signal,
   164→      text: t.koneksiPolling,
   165→    },
   166→    offline: {
   167→      cls: "bg-red-950 text-red-400 border-red-900",
   168→      icon: SignalZero,
   169→      text: "Offline",
   170→    },
   171→  };
   172→  const M = map[mode] ?? map.connecting;
   173→  const Icon = M.icon;
   174→  return (
   175→    <Badge variant="outline" className={`${M.cls} gap-1.5 px-2 py-1 font-medium sm:px-2.5`}>
   176→      <Icon className="h-3.5 w-3.5 shrink-0" />
   177→      <span className="hidden min-w-0 truncate sm:inline">{M.text}</span>
   178→    </Badge>
   179→  );
   180→}

   237→
   238→  // ----- milestone & self-healing toast -----
   239→  const lastMilestoneRef = useRef(0);
   240→  const prevConnMode = useRef<string>("connecting");
   241→  const [recoveryTick, setRecoveryTick] = useState(0);
   242→
   243→  // ----- notifikasi browser (milestone saat tab di background) -----

   246→    try {

   253→      /* abaikan */

   259→      toast.error(t.notifTidakDidukung);

   261→    }
   262→    if (notifEnabled) {
   263→      setNotifEnabled(false);
   264→      try {
   265→        localStorage.setItem("ayam-notif", "0");

   267→      return;
   268→    }

   271→      try {
   272→        perm = await Notification.requestPermission();
   273→      } catch {
   274→        perm = "denied";
   275→      }
   276→    }
   277→    if (perm === "granted") {
   278→      setNotifEnabled(true);
   279→      try {
   280→        localStorage.setItem("ayam-notif", "1");
   281→      } catch {}
   282→      toast.success(t.notifAktif);

   284→      toast.error(t.notifIzinDitolak);

   301→    if (!sessionActive || timeline.length === 0) return 0;
   302→    return timeline[timeline.length - 1]?.t ?? 0;

   307→    const mm = Math.floor((s % 3600) / 60);
   308→    const ss = s % 60;
   309→    const p = (n: number) => String(n).padStart(2, "0");
   310→    return hh > 0 ? `${hh}:${p(mm)}:${p(ss)}` : `${p(mm)}:${p(ss)}`;
   311→  }, [elapsedSec]);

   320→        toast.success(t.backendPulih);
   321→        // Picu sambung-ulang otomatis video feed
   322→        setRecoveryTick((k) => k + 1);

   328→  // ----- beep (WebAudio, tanpa asset) -----
   329→  const playBeep = useCallback(() => {

   335→      const ctx = new Ctx();
   336→      const osc = ctx.createOscillator();
   337→      const gain = ctx.createGain();
   338→      osc.connect(gain);
   339→      gain.connect(ctx.destination);
   340→      osc.type = "sine";
   341→      osc.frequency.value = 880;
   342→      gain.gain.setValueAtTime(0.001, ctx.currentTime);
   343→      gain.gain.exponentialRampToValueAtTime(0.1, ctx.currentTime + 0.02);
   344→      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.45);
   345→      osc.start();
   346→      osc.stop(ctx.currentTime + 0.5);
   347→      osc.onended = () => void ctx.close();
   348→    } catch {
   349→      /* audio diblokir browser — abaikan */
   350→    }
   351→  }, []);
   352→
   353→  // ----- milestone setiap kelipatan 10 -----
   354→  useEffect(() => {
   355→    const c = stats.count;

   357→    if (c < lastMilestoneRef.current) {
   358→      // reset / sesi baru
   359→      lastMilestoneRef.current = c;
   360→      return;
   361→    }
   362→    const prevLevel = Math.floor(lastMilestoneRef.current / STEP);
   363→    const level = Math.floor(c / STEP);
   364→    if (c > 0 && level > prevLevel) {
   365→      const milestoneVal = level * STEP;
   366→      toast.success(`${milestoneVal} ${t.milestoneDesc}`, {
   367→        description: `🎯 ${t.milestone}`,
   368→      });
   369→      playBeep();
   370→      // Notifikasi browser saat tab di background
   371→      if (
   372→        notifEnabled &&
   373→        typeof document !== "undefined" &&
   374→        document.hidden &&
   375→        "Notification" in window &&
   376→        Notification.permission === "granted"
   377→      ) {
   378→        try {
   379→          const n = new Notification(
   380→            `${t.notifMilestoneTitle} — ${milestoneVal}`,
   381→            {
   382→              body: `${milestoneVal} ${t.notifMilestoneBody}`,
   383→              tag: "ayam-milestone",
   384→            }
   385→          );
   386→          setTimeout(() => n.close(), 6000);
   387→        } catch {
   388→          /* abaikan */
   389→        }
   390→      }
   391→    }
   392→    lastMilestoneRef.current = Math.max(lastMilestoneRef.current, c);
   393→  }, [stats.count, t, playBeep, notifEnabled]);
   394→
   395→  // ----- actions -----
   396→  const handleStart = useCallback(async () => {
   397→    if (!canStart) {
   398→      toast.error(t.gagalStart, { description: t.asalAyamRequired });
   399→      return;
   400→    }

   402→    try {
   403→      await ayamApi.startSession({
   404→        asal_ayam: asalAyam.trim(),
   405→        tanggal: tanggal,
   406→        jam: jam,
   407→        keterangan: keterangan.trim(),
   408→      });
   409→      lastMilestoneRef.current = 0;
   410→      toast.success(t.sesiDimulai, {
   411→        description: `${t.asalAyam}: ${asalAyam.trim() || "Unknown"}`,
   412→      });
   413→    } catch {
   414→      toast.error(t.gagalStart, {
   415→        description:
   416→          lang === "id"
   417→            ? "Cek apakah backend Flask berjalan."
   418→            : "Check if the Flask backend is running.",
   419→      });
   420→    } finally {
   421→      setBusy(null);
   422→    }
   423→  }, [asalAyam, tanggal, jam, keterangan, t, lang, canStart]);
   424→
   425→  const handleStop = useCallback(async () => {
   426→    setBusy("stop");

   428→      const res = await ayamApi.stopSession();

   432→              lang === "id" ? "ayam" : "chickens"
   433→            }`
   434→          : lang === "id"
   435→            ? "Tidak ada data deteksi untuk disimpan"

   437→      });
   438→      refreshSideData();
   439→    } catch {
   440→      toast.error(t.gagalStop);
   441→    } finally {
   442→      setBusy(null);
   443→    }
   444→  }, [t, lang, stats.count, refreshSideData]);

   472→    } catch {
   473→      toast.error(t.gagalHapus);

   475→      setDeleteBusy(false);

   479→  const openDetail = useCallback((h: HistoryItem) => {

   481→    setDetailOpen(true);

   484→  // ----- derived -----

   486→
   487→  const weeklyData = useMemo(() => {
   488→    const map = new Map<string, number>();
   489→    for (const h of history.history) {
   490→      if (!h.tanggal) continue;
   491→      map.set(h.tanggal, (map.get(h.tanggal) ?? 0) + (h.total_count ?? 0));

   493→    // Zero-fill: selalu tampilkan 7 hari terakhir (hari tanpa data = 0)

   504→  }, [history]);
   505→

   510→      const okDate = !historyDate || h.tanggal === historyDate;
   511→      return okSearch && okDate;
   512→    });
   513→  }, [history.history, historySearch, historyDate]);
   514→
   515→  // daftar asal ayam unik untuk quick-pick (datalist)
   516→  const originOptions = useMemo(() => {
   517→    const set = new Set<string>();
   518→    for (const h of history.history) {
   519→      const o = (h.asal_ayam ?? "").trim();
   520→      if (o) set.add(o);
   521→    }
   522→    return Array.from(set).slice(0, 12);
   523→  }, [history.history]);
   524→
   525→  const sortedExports = useMemo(
   526→    () => [...exports].sort((a, b) => b.modified - a.modified),
   527→    [exports]
   528→  );
   529→
   530→  const backendBadge =
   531→    device?.backend === "cuda"
   532→      ? "bg-emerald-950 text-emerald-400 border-emerald-900"
   533→      : device?.backend === "cpu"
   534→        ? "bg-amber-950 text-amber-400 border-amber-900"
   535→        : "bg-zinc-800 text-zinc-300 border-zinc-700";
   536→
   537→  return (
   538→    <div className="flex min-h-screen flex-col bg-zinc-950 text-zinc-100">
   539→      {/* ================= HEADER ================= */}
   540→      <header className="sticky top-0 z-50 border-b border-zinc-800 bg-zinc-950/80 backdrop-blur supports-[backdrop-filter]:bg-zinc-950/60">
   541→        <div className="mx-auto flex w-full max-w-7xl items-center justify-between gap-3 px-4 py-3 sm:px-6">
   542→          <div className="flex min-w-0 items-center gap-3">
   543→            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl bg-amber-500 text-zinc-950 shadow-lg shadow-amber-500/20">
   544→              <Bird className="h-6 w-6" />
   545→            </div>
   546→            <div className="min-w-0">
   547→              <h1 className="truncate text-base font-bold leading-tight sm:text-lg">
   548→                {t.appName}
   549→              </h1>
   550→              <p className="hidden truncate text-xs text-zinc-500 sm:block">
   551→                {t.appTagline}
   552→              </p>
   553→            </div>
   554→          </div>
   555→
   556→          <div className="flex items-center gap-2 sm:gap-3">
   557→            <ConnBadge mode={connMode} t={t} />
   558→            <SettingsDialog t={t} onSaved={refreshDevice} />
   559→            <Button
   560→              variant="outline"
   561→              size="icon"
   562→              onClick={() => void toggleNotif()}
   563→              aria-label={notifEnabled ? t.notifNonaktifkan : t.notifAktifkan}
   564→              aria-pressed={notifEnabled}
   565→              title={notifEnabled ? t.notifNonaktifkan : t.notifAktifkan}
