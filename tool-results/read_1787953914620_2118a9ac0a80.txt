   106→def capture_thread():
   107→    """Thread untuk menangkap frame dari camera.
   108→    Mendukung pergantian sumber saat runtime via /api/camera-source:
   109→    - camera_source_request di-set oleh endpoint POST.
   110→    - Loop memeriksa request tiap iterasi; bila ada, buka sumber baru.
   111→    - Bila sumber baru gagal dibuka, sumber lama tetap dipakai dan
   112→      camera_last_error diisi untuk ditampilkan di UI.
   113→    """
   114→    global latest_frame, latest_frame_seq, camera_connected
   115→    global camera_source_request, camera_last_error
   116→
   117→    print("[CAPTURE] Started")
   118→
   119→    cap = None
   120→    current_source = str(Config.CAMERA_SOURCE)
   121→
   122→    def _is_stream(s: str) -> bool:
   123→        return s.startswith(("rtsp://", "http://", "https://"))
   124→
   125→    def _open(source):
   126→        """Coba buka sumber dengan beberapa backend. Return VideoCapture atau None.
   127→        TIDAK me-release cap lama — pemanggil yang mengelola.
   128→        Untuk stream (RTSP/HTTP) lakukan uji baca 1 frame agar sumber mati
   129→        terdeteksi dengan cepat."""
   130→        s = str(source)
   131→        candidates = [0] if s == "0" else [s]
   132→        # Stream (RTSP/HTTP) → coba FFMPEG dulu; file/idx → CAP_ANY dulu
   133→        backends = [cv2.CAP_FFMPEG, cv2.CAP_ANY] if _is_stream(s) else [cv2.CAP_ANY, cv2.CAP_FFMPEG]
   134→        for src in candidates:
   135→            for backend in backends:
   136→                try:
   137→                    c = cv2.VideoCapture(src, backend)
   138→                    if c.isOpened():
   139→                        if _is_stream(s):
   140→                            ret, _ = c.read()
   141→                            if not ret:
   142→                                c.release()
   143→                                print(f"[CAPTURE] Stream opened but no frame: {src}")
   144→                                continue
   145→                            c.set(cv2.CAP_PROP_POS_AVI_RATIO, 0)  # stream: tidak relevan, aman
   146→                        c.set(cv2.CAP_PROP_BUFFERSIZE, 1)
   147→                        c.set(cv2.CAP_PROP_FPS, Config.CAMERA_FPS)
   148→                        c.set(cv2.CAP_PROP_FRAME_WIDTH, Config.CAMERA_WIDTH)
   149→                        c.set(cv2.CAP_PROP_FRAME_HEIGHT, Config.CAMERA_HEIGHT)
   150→                        print(f"[CAPTURE] Connected with backend: {backend} → {src}")
   151→                        return c
   152→                    c.release()
   153→                except Exception as e:
   154→                    print(f"[CAPTURE] Open failed (backend {backend}): {e}")
   155→        return None
   156→
   157→    # ---- Buka sumber awal (retry agar backend tetap hidup saat kamera RTSP belum siap) ----
   158→    while capture_running and cap is None:
   159→        # FIX deadlock: proses permintaan pergantian sumber yang masuk
   160→        # selagi thread masih tertahan di retry sumber awal yang gagal
   161→        # (mis. RTSP mati yang ter-persist di .env saat startup).
   162→        if camera_source_request is not None:
   163→            req = camera_source_request
   164→            camera_source_request = None
   165→            if str(req) != current_source:
   166→                print(f"[CAPTURE] Initial retry: switching source → {req}")
   167→                current_source = str(req)
   168→        cap = _open(current_source)
   169→        if cap is not None:
   170→            break
   171→        camera_connected = False
   172→        camera_last_error = f"Gagal membuka sumber: {current_source} — mencoba ulang..."
   173→        print(f"[CAPTURE] Camera failed to open, retry in 3s: {current_source}")
   174→        time.sleep(3)
   175→
   176→    is_file_source = not _is_stream(current_source) and current_source != "0"
   177→    frame_interval = 1.0 / max(Config.CAMERA_FPS, 1)
   178→
   179→    print(f"[CAPTURE] Camera opened: {current_source}")
   180→    print(f"[CAPTURE] Resolution: {Config.CAMERA_WIDTH}x{Config.CAMERA_HEIGHT}")
   181→    print(f"[CAPTURE] FPS: {Config.CAMERA_FPS}")
   182→
   183→    camera_connected = True
   184→    camera_last_error = None
   185→
   186→    while capture_running:
   187→        # ---- Pergantian sumber saat runtime ----
   188→        if camera_source_request is not None:
   189→            req = camera_source_request
   190→            camera_source_request = None
   191→            if str(req) != current_source:
   192→                print(f"[CAPTURE] Switching source → {req}")
   193→                new_cap = _open(req)
   194→                if new_cap is not None:
   195→                    try:
   196→                        cap.release()
   197→                    except Exception:
   198→                        pass
   199→                    cap = new_cap
   200→                    current_source = str(req)
   201→                    is_file_source = not _is_stream(current_source) and current_source != "0"
   202→                    camera_connected = True
   203→                    camera_last_error = None
   204→                    print(f"[CAPTURE] Source switched OK: {current_source}")
   205→                else:
   206→                    camera_last_error = f"Gagal membuka sumber baru: {req} — memakai sumber lama"
   207→                    print(f"[CAPTURE] {camera_last_error}")
   208→                    # Pulihkan Config + .env ke sumber lama agar konsisten dengan
   209→                    # yang benar-benar diputar di layar
   210→                    try:
   211→                        Config.CAMERA_SOURCE = current_source
   212→                        _update_env_file({"CAMERA_SOURCE": current_source})
   213→                    except Exception:
   214→                        pass
   215→            continue
   216→
   217→        ret, frame = cap.read()
   218→
   219→        if not ret:
   220→            # Untuk file video: loop ulang dari awal (mode demo/sandbox)
   221→            if is_file_source:
   222→                print("[CAPTURE] Video ended, restarting from beginning...")
   223→                cap.set(cv2.CAP_PROP_POS_FRAMES, 0)
   224→                time.sleep(0.05)
   225→                continue
   226→            print("[CAPTURE] Failed to read frame, reconnecting...")
   227→            camera_connected = False
   228→            cap.release()
   229→            time.sleep(2)
   230→            cap = _open(current_source)
   231→            if cap is not None:
   232→                camera_connected = True
   233→                camera_last_error = None
   234→            else:
   235→                camera_last_error = f"Koneksi kamera terputus: {current_source}"
   236→            continue
   237→        
   238→        # Pacing untuk file video agar tidak membaca secepat decode
   239→        if is_file_source:
   240→            time.sleep(frame_interval)
   241→        
   242→        with frame_lock:
   243→            latest_frame = frame.copy()
   244→            latest_frame_seq += 1
   245→        
   246→        if is_processing:
   247→            try:
   248→                frame_queue.put_nowait(frame)
   249→            except queue.Full:
   250→                # Drop-oldest policy: buang frame terlama, masukkan frame terbaru
   251→                # sehingga queue selalu berisi frame paling baru (akurasi crossing)
   252→                try:
   253→                    frame_queue.get_nowait()
   254→                except queue.Empty:
   255→                    pass
   256→                frame_queue.put_nowait(frame)
   257→    
   258→    cap.release()
   259→    camera_connected = False
   260→    print("[CAPTURE] Stopped")

   266→def detection_thread():
   267→    """Thread untuk menjalankan deteksi dan penghitungan"""
   268→    global current_count, current_speed, current_tracks, latest_detections
   269→    global latest_display_frame, latest_display_seq
   270→    
   271→    print("[DETECTION] Started")
   272→    frame_count = 0
   273→    last_emit_time = time.time()
   274→    
   275→    while detection_running:
   276→        try:
   277→            frame = frame_queue.get(timeout=1)
   278→            frame_count += 1
   279→            
   280→            # ============================================
   281→            # DETEKSI
   282→            # ============================================
   283→            detections = detector.detect(frame)
   284→            
   285→            # ============================================
   286→            # HITUNG DENGAN SIMPLE COUNTER
   287→            # ============================================
   288→            count = counter.update(detections)
   289→            
   290→            # Update global variables
   291→            current_count = count
   292→            current_speed = 0  # Tidak ada speed tracking
   293→            current_tracks = len([d for d in detections if d.get("is_chicken", False)])
   294→            
   295→            # ============================================
   296→            # FILTER DETEKSI YANG SUDAH DIHITUNG
   297→            # ============================================
   298→            # Hapus deteksi yang sudah dihitung dari tampilan
   299→            filtered_detections = []
   300→            for d in detections:
   301→                if not d.get("counted", False):
   302→                    filtered_detections.append(d)
   303→            
   304→            # Simpan untuk ditampilkan
   305→            with detection_lock:
   306→                latest_detections = filtered_detections.copy()
   307→            
   308→            # ============================================
   309→            # GAMBAR VISUALISASI PADA FRAME YANG DIDETEKSI
   310→            # ============================================
   311→            # Overlay digambar di sini (frame deteksi + deteksi yang sama),
   312→            # sehingga feed tidak mencampur dua frame berbeda (Fix 3).
   313→            display_frame = draw_visualization(
   314→                frame,
   315→                filtered_detections,
   316→                counter.count_line_x,
   317→                count
   318→            )
   319→            with detection_lock:
   320→                latest_display_frame = display_frame
   321→                latest_display_seq += 1
   322→            
   323→            # ============================================
   324→            # SAVE KE EXCEL
   325→            # ============================================
   326→            if session_active and count > 0:
   327→                # Simpan setiap penambahan count (dedup otomatis oleh exporter)
   328→                for d in detections:
   329→                    if d.get("is_chicken", False):
   330→                        excel_exporter.add_detection_direct(
   331→                            count, 0, d
   332→                        )
   333→                        break
   334→            
   335→            # ============================================
   336→            # EMIT STATS KE CLIENT
   337→            # ============================================
   338→            if time.time() - last_emit_time > 0.3:
   339→                socketio.emit("update_stats", {
   340→                    "count": count,
   341→                    "speed": 0,
   342→                    "tracks": current_tracks,
   343→                    "timestamp": time.time(),
   344→                    "session_active": session_active,
   345→                    "session_data": session_data,
   346→                    "method": "Simple Counter",
   347→                    "frame": frame_count
   348→                })
   349→                last_emit_time = time.time()
   350→            
   351→        except queue.Empty:
   352→            continue
   353→        except Exception as e:
   354→            print(f"[DETECTION ERROR] {e}")

   360→def draw_visualization(frame, detections, count_line_x, total_count):
   361→    """
   362→    Gambar visualisasi dengan garis di kiri
   363→    """
   364→    height, width = frame.shape[:2]
   365→    
   366→    # ============================================
   367→    # 1. GARIS MERAH DI KIRI (VERTIKAL)
   368→    # ============================================
   369→    cv2.line(frame, (count_line_x, 0), (count_line_x, height), (0, 0, 255), 3)
   370→    
   371→    # Label garis
   372→    cv2.putText(frame, "COUNT LINE", (count_line_x - 60, 30),
   373→               cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 0, 255), 2)
   374→    
   375→    # ============================================
   376→    # 2. ZONE DETEKSI (KUNING)
   377→    # ============================================
   378→    zone_start = count_line_x - counter.zone_width // 2
   379→    zone_end = count_line_x + counter.zone_width // 2
   380→    cv2.rectangle(frame, (zone_start, 0), (zone_end, height), (0, 255, 255), 1)
   381→    
   382→    # ============================================
   383→    # 3. GAMBAR DETEKSI
   384→    # ============================================
   385→    for det in detections:
   386→        if det.get("is_chicken", False):
   387→            x1, y1, x2, y2 = det["x1"], det["y1"], det["x2"], det["y2"]
   388→            cx, cy = det["center_x"], det["center_y"]
   389→            conf = det["confidence"]
   390→            
   391→            # Cek apakah sudah dihitung (tidak akan muncul karena sudah difilter)
   392→            if det.get("counted", False):
   393→                color = (0, 255, 0)  # Hijau - sudah dihitung
   394→                label = f"✓ {conf:.2f}"
   395→            else:
   396→                color = (0, 255, 255)  # Kuning - belum dihitung
   397→                label = f"? {conf:.2f}"
   398→            
   399→            # Gambar bounding box
   400→            cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
   401→            
   402→            # Gambar center point
   403→            cv2.circle(frame, (cx, cy), 4, (255, 0, 0), -1)
   404→            
   405→            # Label
   406→            cv2.putText(frame, label, (x1, y1-10),
   407→                       cv2.FONT_HERSHEY_SIMPLEX, 0.4, color, 1)
   408→    
   409→    # ============================================
   410→    # 4. TOTAL COUNTER
   411→    # ============================================
   412→    cv2.putText(frame, f"TOTAL: {total_count}", (10, 60),
   413→               cv2.FONT_HERSHEY_SIMPLEX, 1.2, (0, 255, 0), 3)
   414→    
   415→    # ============================================
   416→    # 5. INFORMASI TAMBAHAN
   417→    # ============================================
   418→    # Jumlah ayam yang terdeteksi
   419→    total_detected = len([d for d in detections if d.get("is_chicken", False)])
   420→    
   421→    cv2.putText(frame, f"DETEKSI: {total_detected}", (10, 90),
   422→               cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)
   423→    
   424→    # Status session
   425→    status = "ACTIVE" if session_active else "INACTIVE"
   426→    color = (0, 255, 0) if session_active else (0, 0, 255)
   427→    cv2.putText(frame, f"SESSION: {status}", (10, 120),
   428→               cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)
   429→    
   430→    # ============================================
   431→    # 6. PANAH PETUNJUK
   432→    # ============================================
   433→    cv2.putText(frame, "--> COUNT HERE", (count_line_x + 30, height - 20),
   434→               cv2.FONT_HERSHEY_SIMPLEX, 0.4, (0, 0, 255), 1)
   435→    
   436→    return frame

   438→def generate_frames():
   439→    """Generate video frames with overlay (paced, hanya yield frame baru)"""
   440→    frame_count = 0
   441→    last_seq = -1
   442→    last_yield_time = 0.0
   443→    target_interval = 1.0 / max(Config.CAMERA_FPS, 1)
   444→
   445→    while True:
   446→        processing = is_processing
   447→
   448→        # Saat processing: pakai frame display hasil deteksi (sinkron, Fix 3).
   449→        # Saat idle: pakai frame mentah dari kamera agar feed tetap tampil.
   450→        if processing:
   451→            with detection_lock:
   452→                seq = latest_display_seq
   453→                frame = latest_display_frame.copy() if latest_display_frame is not None else None
   454→        else:
   455→            with frame_lock:
   456→                seq = latest_frame_seq
   457→                frame = latest_frame.copy() if latest_frame is not None else None
   458→
   459→        if frame is None:
   460→            time.sleep(0.05)
   461→            continue
   462→
   463→        # ============================================
   464→        # PACING: hanya yield jika ada frame baru (seq berubah)
   465→        # ============================================
   466→        # Mencegah encode duplikat frame yang sama (kurangi beban CPU).
   467→        if seq == last_seq:
   468→            time.sleep(0.02)
   469→            continue
   470→
   471→        # Pace output ke interval stabil agar tidak burst
   472→        now = time.time()
   473→        wait = target_interval - (now - last_yield_time)
   474→        if wait > 0:
   475→            time.sleep(wait)
   476→
   477→        # Baca ulang frame terbaru setelah pacing.
   478→        # Selalu tampilkan frame PALING BARU (tidak skip) agar feed tidak
   479→        # berhenti ketika producer menghasilkan frame secepat target interval.
   480→        if processing:
   481→            with detection_lock:
   482→                seq = latest_display_seq
   483→                frame = latest_display_frame.copy() if latest_display_frame is not None else None
   484→        else:
   485→            with frame_lock:
   486→                seq = latest_frame_seq
   487→                frame = latest_frame.copy() if latest_frame is not None else None
   488→            if frame is not None:
   489→                # Idle: gambar visualisasi tanpa deteksi
   490→                frame = draw_visualization(
   491→                    frame,
   492→                    [],
   493→                    counter.count_line_x,
   494→                    current_count
   495→                )
   496→
   497→        if frame is None:
   498→            time.sleep(0.02)
   499→            continue
   500→
   501→        last_seq = seq
   502→        last_yield_time = time.time()
   503→        frame_count += 1
   504→
   505→        # Frame counter
   506→        height, width = frame.shape[:2]
   507→        cv2.putText(frame, f"FRAME: {frame_count}", (width-150, 30),
   508→                   cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 1)
   509→        
   510→        # ============================================
   511→        # ENCODE DAN YIELD
   512→        # ============================================
   513→        ret, buffer = cv2.imencode(".jpg", frame, [cv2.IMWRITE_JPEG_QUALITY, 80])
   514→        yield (b"--frame\r\n"
   515→               b"Content-Type: image/jpeg\r\n\r\n" +
   516→               buffer.tobytes() +
   517→               b"\r\n")

   759→def _drain_frame_queue():
   760→    """Buang semua frame sisa di queue agar tidak terhitung saat sesi berikutnya"""
   761→    drained = 0
   762→    while True:
   763→        try:
   764→            frame_queue.get_nowait()
   765→            drained += 1
   766→        except queue.Empty:
   767→            break
   768→    if drained:
   769→        print(f"[QUEUE] Drained {drained} stale frames")

   771→def _do_start_session(data):
   772→    """Logika start session (dipakai SocketIO + REST)"""
   773→    global is_processing, session_active, session_data, current_count, current_tracks, session_started_at
   774→
   775→    # GUARD: jika masih ada sesi aktif, simpan dulu sesi lama agar datanya
   776→    # tidak hilang (bug lama: start baru menimpa buffer Excel tanpa disimpan).
   777→    if session_active:
   778→        print("[SESSION] Overwrite guard: previous session still active — auto-saving it first")
   779→        try:
   780→            _do_stop_session()
   781→        except Exception as e:
   782→            print(f"[SESSION] Auto-save of previous session failed: {e}")
   783→
   784→    _drain_frame_queue()
   785→    counter.reset()
   786→    current_count = 0
   787→    current_tracks = 0
   788→    
   789→    session_data = {
   790→        "asal_ayam": data.get("asal_ayam", "Unknown"),
   791→        "tanggal": data.get("tanggal", datetime.now().strftime("%Y-%m-%d")),
   792→        "jam": data.get("jam", datetime.now().strftime("%H:%M")),
   793→        "keterangan": data.get("keterangan", "")
   794→    }
   795→    
   796→    excel_exporter.start_new_session(
   797→        session_data["asal_ayam"],
   798→        session_data["jam"],
   799→        session_data["tanggal"],
   800→        session_data["keterangan"]
   801→    )
   802→    
   803→    session_started_at = datetime.now().isoformat()
   804→    session_active = True
   805→    is_processing = True
   806→    print(f"[SESSION] Started: {session_data}")
   807→
   808→def _do_stop_session():
   809→    """Logika stop session (dipakai SocketIO + REST). Simpan Excel + DB"""
   810→    global is_processing, session_active
   811→    
   812→    is_processing = False
   813→    session_active = False
   814→    _drain_frame_queue()
   815→    
   816→    saved_file = excel_exporter.stop_and_save()
   817→    
   818→    # Snapshot timeline kumulatif [{t: detik, total: n}, ...] ke DB agar
   819→    # grafik pertumbuhan sesi tetap tampil di dialog detail setelah restart
   820→    # backend (dulu hanya hidup di memori counter).
   821→    timeline_json = None
   822→    try:
   823→        start_ts = None
   824→        if session_started_at:
   825→            try:
   826→                start_ts = datetime.fromisoformat(session_started_at).timestamp()
   827→            except Exception:
   828→                start_ts = None
   829→        points = []
   830→        for h in getattr(counter, "count_history", []):
   831→            t = (h["timestamp"] - start_ts) if start_ts else 0.0
   832→            points.append({"t": round(max(t, 0.0), 1), "total": int(h["count"])})
   833→        if points:
   834→            timeline_json = json.dumps(points)
   835→    except Exception as e:
   836→        print(f"[SESSION] Timeline snapshot failed: {e}")
   837→    
   838→    # Simpan riwayat session ke database agar history berfungsi
   839→    try:
   840→        db.add_session(
   841→            asal_ayam=session_data.get("asal_ayam", "Unknown"),
   842→            tanggal=session_data.get("tanggal", ""),
   843→            jam=session_data.get("jam", ""),
   844→            keterangan=session_data.get("keterangan", ""),
   845→            total_count=current_count,
   846→            file_name=saved_file or "",
   847→            start_time=session_started_at,
   848→            timeline_json=timeline_json,
   849→        )
   850→    except Exception as e:
   851→        print(f"[DB ERROR] Failed to save session history: {e}")
   852→    
   853→    print(f"[SESSION] Stopped, file saved: {saved_file}")
   854→    return saved_file

   939→def _update_env_file(updates: dict):
   940→    """Perbarui key=value di file .env (tambahkan bila key belum ada)"""
   941→    lines = []
   942→    if os.path.exists(ENV_PATH):
   943→        with open(ENV_PATH, "r", encoding="utf-8") as f:
   944→            lines = f.readlines()
   945→    pending = dict(updates)
   946→    for i, line in enumerate(lines):
   947→        m = re.match(r"^([A-Z_]+)\s*=", line)
   948→        if m and m.group(1) in pending:
   949→            key = m.group(1)
   950→            lines[i] = f"{key}={pending.pop(key)}\n"
   951→    for key, val in pending.items():
   952→        if lines and not lines[-1].endswith("\n"):
   953→            lines.append("\n")
   954→        lines.append(f"{key}={val}\n")
   955→    with open(ENV_PATH, "w", encoding="utf-8") as f:
   956→        f.writelines(lines)
   957→    print(f"[SETTINGS] .env updated: {updates}")

  1244→def _durasi_str(s):
  1245→    """Durasi sesi 'Xj Ym' / 'Ym Zs' / 'Zs' dari start_time & end_time ISO."""
  1246→    try:
  1247→        st = datetime.fromisoformat(s["start_time"])
  1248→        en = datetime.fromisoformat(s["end_time"])
  1249→        secs = max(0, int((en - st).total_seconds()))
  1250→        if secs <= 0:
  1251→            return "-"
  1252→        m, sec = divmod(secs, 60)
  1253→        h, m = divmod(m, 60)
  1254→        return f"{h}j {m}m" if h else (f"{m}m {sec:02d}s" if m else f"{sec}s")
  1255→    except Exception:
  1256→        return "-"

  1259→def _rl_bar_chart(bars, empty_text, bar_color="#f59e0b", width=500, height=110, label_size=5.6):
  1260→    """Grafik batang reportlab sederhana. bars: list of (label, value)."""
  1261→    d = Drawing(width, height)
  1262→    if not bars:
  1263→        d.add(String(20, 60, empty_text, fontName="Helvetica-Oblique", fontSize=9,
  1264→                     fillColor=HexColor("#a1a1aa")))
  1265→        return d
  1266→    area_w, area_h = width - 30, height - 30
  1267→    max_c = max(1, max(v for _, v in bars))
  1268→    n = len(bars)
  1269→    bar_w = min(28.0, max(4.0, area_w / max(n, 1) * 0.6))
  1270→    step = area_w / max(n, 1)
  1271→    d.add(Rect(20, 14, area_w, 0.7, fillColor=HexColor("#d4d4d8"), strokeColor=None))
  1272→    for i, (lbl_txt, v) in enumerate(bars[:48]):  # maks 48 bar agar tetap terbaca
  1273→        bh = (v / max_c) * area_h
  1274→        x = 20 + i * step + (step - bar_w) / 2
  1275→        d.add(Rect(x, 15, bar_w, max(bh, 0.8), fillColor=HexColor(bar_color), strokeColor=None))
  1276→        lbl = String(x + bar_w / 2, 5, str(lbl_txt), fontName="Helvetica",
  1277→                     fontSize=label_size, fillColor=HexColor("#71717a"))
  1278→        lbl.textAnchor = "middle"
  1279→        d.add(lbl)
  1280→        if v > 0:
  1281→            vlab = String(x + bar_w / 2, 15 + bh + 2.5, str(v), fontName="Helvetica-Bold",
  1282→                          fontSize=label_size + 0.4, fillColor=HexColor("#b45309"))
  1283→            vlab.textAnchor = "middle"
  1284→            d.add(vlab)
  1285→    return d
