   145→    string,

   151→      text: `${t.koneksiMode}: …`,
   152→    },

   164→      cls: "bg-red-950 text-red-400 border-red-900",
   165→      icon: SignalZero,
   166→      text: "Offline",

   168→  };
   169→  const M = map[mode] ?? map.connecting;
   170→  const Icon = M.icon;
   171→  return (
   172→    <Badge variant="outline" className={`${M.cls} gap-1.5 px-2 py-1 font-medium sm:px-2.5`}>

   175→    </Badge>
   176→  );
   177→}

   183→export default function AyamCounterPage() {
   184→  const [lang, setLangState] = useState<Lang>("id");
   185→
   186→  // Muat bahasa tersimpan (client only, hindari hydration mismatch)
   187→  useEffect(() => {
   188→    try {
   189→      const saved = window.localStorage.getItem("ayam-lang");

   192→      /* abaikan */
   193→    }
   194→  }, []);
   195→
   196→  const setLang = useCallback((l: Lang) => {
   197→    setLangState(l);
   198→    try {
   199→      window.localStorage.setItem("ayam-lang", l);
   200→    } catch {
   201→      /* abaikan */
   202→    }
   203→  }, []);
   204→
   205→  const t = dict[lang];
   206→
   207→  const {
   208→    stats,
   209→    device,
   210→    history,
   211→    exports,
   212→    timeline,
   213→    connMode,
   214→    refreshSideData,
   215→    refreshDevice,
   216→  } = useAyamDashboard();
   217→
   218→  // ----- form state -----
   219→  const [asalAyam, setAsalAyam] = useState("");
   220→  const [tanggal, setTanggal] = useState("");
   221→  const [jam, setJam] = useState("");
   222→  const [keterangan, setKeterangan] = useState("");

   224→
   225→  // ----- history filter -----
   226→  const [historySearch, setHistorySearch] = useState("");
   227→  const [historyDate, setHistoryDate] = useState("");
   228→
   229→  // ----- detail & delete dialog -----
   230→  const [detailSession, setDetailSession] = useState<HistoryItem | null>(null);
   231→  const [detailOpen, setDetailOpen] = useState(false);
   232→  const [deleteTarget, setDeleteTarget] = useState<HistoryItem | null>(null);
   233→  const [deleteBusy, setDeleteBusy] = useState(false);
   234→
   235→  // ----- milestone & self-healing toast -----
   236→  const lastMilestoneRef = useRef(0);
   237→  const prevConnMode = useRef<string>("connecting");
   238→  const [recoveryTick, setRecoveryTick] = useState(0);
   239→
   240→  // default tanggal & jam (client only, hindari hydration mismatch)
   241→  useEffect(() => {
   242→    const now = new Date();
   243→    const pad = (n: number) => String(n).padStart(2, "0");
   244→    setTanggal(`${now.getFullYear()}-${pad(now.getMonth() + 1)}-${pad(now.getDate())}`);

   246→  }, []);
   247→
   248→  const sessionActive = stats.session_active || stats.is_processing === true;
   249→  const canStart = asalAyam.trim().length > 0;
   250→
   251→  // ----- durasi sesi berjalan (dari timeline backend, detik) -----
   252→  const elapsedSec = useMemo(() => {
   253→    if (!sessionActive || timeline.length === 0) return 0;
   254→    return timeline[timeline.length - 1]?.t ?? 0;
   255→  }, [sessionActive, timeline]);
   256→  const elapsedLabel = useMemo(() => {
   257→    const s = Math.max(0, Math.floor(elapsedSec));
   258→    const hh = Math.floor(s / 3600);
   259→    const mm = Math.floor((s % 3600) / 60);
   260→    const ss = s % 60;
   261→    const p = (n: number) => String(n).padStart(2, "0");
   262→    return hh > 0 ? `${hh}:${p(mm)}:${p(ss)}` : `${p(mm)}:${p(ss)}`;
   263→  }, [elapsedSec]);
   264→
   265→  // ----- toast saat backend down / pulih (self-healing) -----
   266→  useEffect(() => {
   267→    const prev = prevConnMode.current;
   268→    if (prev !== connMode) {
   269→      if (connMode === "offline") {
   270→        toast.info(t.backendMulaiUlang, { duration: 8000 });
   271→      } else if (prev === "offline" && (connMode === "socket" || connMode === "polling")) {
   272→        toast.success(t.backendPulih);
   273→        // Picu sambung-ulang otomatis video feed
   274→        setRecoveryTick((k) => k + 1);
   275→      }
   276→      prevConnMode.current = connMode;
   277→    }
   278→  }, [connMode, t]);
   279→
   280→  // ----- beep (WebAudio, tanpa asset) -----
   281→  const playBeep = useCallback(() => {
   282→    try {
   283→      const Ctx =
   284→        window.AudioContext ||
   285→        (window as unknown as { webkitAudioContext: typeof AudioContext })
   286→          .webkitAudioContext;
   287→      const ctx = new Ctx();
   288→      const osc = ctx.createOscillator();
   289→      const gain = ctx.createGain();
   290→      osc.connect(gain);
   291→      gain.connect(ctx.destination);
   292→      osc.type = "sine";
   293→      osc.frequency.value = 880;
   294→      gain.gain.setValueAtTime(0.001, ctx.currentTime);
   295→      gain.gain.exponentialRampToValueAtTime(0.1, ctx.currentTime + 0.02);
   296→      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.45);
   297→      osc.start();
   298→      osc.stop(ctx.currentTime + 0.5);
   299→      osc.onended = () => void ctx.close();
   300→    } catch {
   301→      /* audio diblokir browser — abaikan */
   302→    }
   303→  }, []);

   305→  // ----- milestone setiap kelipatan 10 -----

   312→      return;

   315→    const level = Math.floor(c / STEP);
   316→    if (c > 0 && level > prevLevel) {
   317→      const milestoneVal = level * STEP;
   318→      toast.success(`${milestoneVal} ${t.milestoneDesc}`, {
   319→        description: `🎯 ${t.milestone}`,
   320→      });
   321→      playBeep();
   322→    }
   323→    lastMilestoneRef.current = Math.max(lastMilestoneRef.current, c);
   324→  }, [stats.count, t, playBeep]);
   325→

   327→  const handleStart = useCallback(async () => {
   328→    if (!canStart) {
   329→      toast.error(t.gagalStart, { description: t.asalAyamRequired });
   330→      return;
   331→    }
   332→    setBusy("start");
   333→    try {
   334→      await ayamApi.startSession({
   335→        asal_ayam: asalAyam.trim(),
   336→        tanggal: tanggal,
   337→        jam: jam,
   338→        keterangan: keterangan.trim(),
   339→      });
   340→      lastMilestoneRef.current = 0;
   341→      toast.success(t.sesiDimulai, {
   342→        description: `${t.asalAyam}: ${asalAyam.trim() || "Unknown"}`,
   343→      });
   344→    } catch {
   345→      toast.error(t.gagalStart, {
   346→        description:
   347→          lang === "id"
   348→            ? "Cek apakah backend Flask berjalan."
   349→            : "Check if the Flask backend is running.",
   350→      });
   351→    } finally {
   352→      setBusy(null);
   353→    }
   354→  }, [asalAyam, tanggal, jam, keterangan, t, lang, canStart]);
   355→
   356→  const handleStop = useCallback(async () => {
   357→    setBusy("stop");
   358→    try {
   359→      const res = await ayamApi.stopSession();
   360→      toast.success(t.sesiDihentikan, {
   361→        description: res.file
   362→          ? `${res.file.split("/").pop()} — ${stats.count} ${
   363→              lang === "id" ? "ayam" : "chickens"
   364→            }`
   365→          : lang === "id"
   366→            ? "Tidak ada data deteksi untuk disimpan"
   367→            : "No detection data to save",
   368→      });
   369→      refreshSideData();
   370→    } catch {
   371→      toast.error(t.gagalStop);
   372→    } finally {
   373→      setBusy(null);
   374→    }
   375→  }, [t, lang, stats.count, refreshSideData]);
   376→
   377→  const handleReset = useCallback(async () => {
   378→    setBusy("reset");
   379→    try {
   380→      await ayamApi.resetCounter();
   381→      lastMilestoneRef.current = 0;
   382→      toast.info(t.counterDireset);
   383→    } catch {
   384→      toast.error(t.gagalReset);
   385→    } finally {
   386→      setBusy(null);
   387→    }
   388→  }, [t]);
   389→
   390→  // ----- hapus sesi riwayat -----
   391→  const handleDelete = useCallback(async () => {
   392→    if (!deleteTarget) return;
   393→    setDeleteBusy(true);
   394→    try {
   395→      const res = await ayamApi.deleteSession(deleteTarget.id);
   396→      toast.success(t.sesiDihapus, {
   397→        description: res.file_removed
   398→          ? `${deleteTarget.file_name?.split("/").pop()} — ✓`
   399→          : `#${deleteTarget.id}`,
   400→      });
   401→      setDeleteTarget(null);
   402→      refreshSideData();
   403→    } catch {
   404→      toast.error(t.gagalHapus);
   405→    } finally {
   406→      setDeleteBusy(false);
   407→    }
   408→  }, [deleteTarget, t, refreshSideData]);
   409→
   410→  const openDetail = useCallback((h: HistoryItem) => {
   411→    setDetailSession(h);
   412→    setDetailOpen(true);
   413→  }, []);
   414→
   415→  // ----- derived -----
   416→  const daily = history.stats;
   417→
   418→  const weeklyData = useMemo(() => {

   420→    for (const h of history.history) {

   441→  }, [history.history, historySearch, historyDate]);

   453→  const sortedExports = useMemo(
   454→    () => [...exports].sort((a, b) => b.modified - a.modified),
   455→    [exports]
   456→  );
   457→
   458→  const backendBadge =
   459→    device?.backend === "cuda"
   460→      ? "bg-emerald-950 text-emerald-400 border-emerald-900"
   461→      : device?.backend === "cpu"
   462→        ? "bg-amber-950 text-amber-400 border-amber-900"
   463→        : "bg-zinc-800 text-zinc-300 border-zinc-700";
   464→
   465→  return (
   466→    <div className="flex min-h-screen flex-col bg-zinc-950 text-zinc-100">
   467→      {/* ================= HEADER ================= */}
   468→      <header className="sticky top-0 z-50 border-b border-zinc-800 bg-zinc-950/80 backdrop-blur supports-[backdrop-filter]:bg-zinc-950/60">
   469→        <div className="mx-auto flex w-full max-w-7xl items-center justify-between gap-3 px-4 py-3 sm:px-6">
   470→          <div className="flex min-w-0 items-center gap-3">
   471→            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl bg-amber-500 text-zinc-950 shadow-lg shadow-amber-500/20">
   472→              <Bird className="h-6 w-6" />
   473→            </div>
   474→            <div className="min-w-0">
   475→              <h1 className="truncate text-base font-bold leading-tight sm:text-lg">
   476→                {t.appName}
   477→              </h1>
   478→              <p className="hidden truncate text-xs text-zinc-500 sm:block">
   479→                {t.appTagline}
   480→              </p>
   481→            </div>
   482→          </div>
   483→
   484→          <div className="flex items-center gap-2 sm:gap-3">

   488→            {/* Language toggle */}

   490→              <button
   491→                onClick={() => setLang("id")}
   492→                className={`px-3 py-1.5 transition-colors ${
   493→                  lang === "id"
   494→                    ? "bg-amber-500 text-zinc-950"
   495→                    : "text-zinc-400 hover:text-zinc-200"
   496→                }`}
   497→                aria-pressed={lang === "id"}
   498→              >
   499→                ID
   500→              </button>
   501→              <button
   502→                onClick={() => setLang("en")}
   503→                className={`px-3 py-1.5 transition-colors ${
   504→                  lang === "en"
   505→                    ? "bg-amber-500 text-zinc-950"
   506→                    : "text-zinc-400 hover:text-zinc-200"
   507→                }`}
   508→                aria-pressed={lang === "en"}
   509→              >
   510→                EN
   511→              </button>
   512→            </div>
   513→          </div>
   514→        </div>
   515→      </header>
   516→
   517→      {/* ================= MAIN ================= */}
   518→      <main className="mx-auto w-full max-w-7xl flex-1 px-4 py-5 sm:px-6 sm:py-6">
   519→        {/* ---- Stat cards ---- */}
   520→        <motion.section
   521→          aria-label="statistics"
   522→          initial={fadeUp.initial}
   523→          animate={fadeUp.animate}
   524→          transition={{ duration: 0.35 }}
   525→          className="grid grid-cols-2 gap-3 sm:gap-4 xl:grid-cols-4"
   526→        >
   527→          <StatCard
   528→            icon={Bird}
   529→            label={t.totalAyam}
   530→            value={
   531→              <AnimatedNumber
   532→                value={stats.count}
   533→                className={sessionActive ? "text-emerald-400" : undefined}
   534→              />
   535→            }
   536→            sub={
   537→              sessionActive ? (
   538→                <span className="inline-flex items-center gap-1.5 font-medium text-emerald-400">
   539→                  <LiveDot active /> {t.aktifBerjalan}
   540→                </span>
   541→              ) : (
   542→                <span className="inline-flex items-center gap-1.5 text-zinc-500">
   543→                  <LiveDot active={false} /> {t.menungguSesi}
   544→                </span>
   545→              )
   546→            }
   547→            accent="bg-amber-500/15 text-amber-400"
   548→            glow="hover:shadow-[0_8px_32px_-12px_rgba(245,158,11,0.35)]"
   549→          />
   550→          <StatCard
   551→            icon={ScanEye}
